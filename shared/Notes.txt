🧩 كود Register كامل
import { addUser } from "./services/userService.js";

// 🔹 Register function
async function register() {
  const name = document.getElementById("name").value;
  const email = document.getElementById("email").value;
  const password = document.getElementById("password").value;
  const type = document.getElementById("type").value;

  // 1️⃣ Create user object
  const userData = {
    name,
    email,
    password,
    user_type: type
  };

  // 2️⃣ Add user to users table
  const result = await addUser(userData);

  if (!result.success) {
    alert(result.message);
    return;
  }

  const userId = result.data.id;

  // 3️⃣ حسب النوع
  if (type === "player") {
    await registerPlayer(userId);
  }

  if (type === "specialist") {
    await registerSpecialist(userId);
  }

  alert("User registered successfully 🎉");
}
🟢 Player Register
async function registerPlayer(userId) {
  const playerData = {
    userId: userId,
    height: 170,
    weight: 65,
    level: "beginner"
  };

  await fetch(`https://your-firebase-url/players/${userId}.json`, {
    method: "PUT",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify(playerData)
  });
}
🔵 Specialist Register
async function registerSpecialist(userId) {
  const specialistData = {
    userId: userId,
    specialization: "orthopedic",
    experienceYears: 5,
    clinicId: "c1"
  };

  await fetch(`https://your-firebase-url/specialists/${userId}.json`, {
    method: "PUT",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify(specialistData)
  });
}
